## 重要 scope 修订（2026-09-02 doc audit reframe）

**旧 plan doc 假定编译器 emit raw x64（prologue / epilogue / shadow space / register alloc 都想抽到 ABI 文件）。这跟实际 architecture 不符**：JHYY 编译器 emit QBE IL，**所有 x64 细节由 QBE 处理**：
- register 分配 (rcx/rdx/r8/r9) — QBE `-t amd64_win`
- shadow space (32 字节) — QBE
- frame layout（prologue/epilogue）— QBE
- callee-saved 寄存器保存恢复 — QBE

grep 验证：`compute_locals_size` / `frame_size` / `locals_size` 在 `codegen.jhyy` 和 `codegen.c` 各 **0 命中**；`push rbp` / `mov rbp` / `sub rsp` / `rcx` / `rdx` / `r8` / `r9` 各 **0 命中**。**jhyy-side 不能抽 x64 prologue/epilogue/shadow space（这些都不存在）**。

**实际可抽的 ABI 表面（QBE-level）**:
1. **QBE function signature 构造** — `export function w $name(l %ret, w %a, w %b) {` + `@start` 标签
2. **sret 预处理** — struct 返回 → 隐藏 `l %ret` 前置 + sret slot alloc + 返回前 copy_struct
3. **struct-arg 拷贝到 slot** — struct 实参 → `alloc8 <bytes>` + `cg_copy_struct` 拷贝 + 传 slot 地址
4. **type coercion at call site** — `cg_convert_arg` (dtosi/stosi/extsw/extuw/sltof/truncd QBE ops)

总 ABI-shaped 代码 ~1150 LOC (jhyy) + ~660 LOC (C-side mirror)，Stage 1 抽到 `abi_amd64_win.jhyy` / `abi_amd64_win.c`。

## Stage 1: abi_amd64_win 抽离（QBE-level）

### 1.1 范围

#### 1.1.1 新建 `abi_amd64_win.jhyy`（5 个函数）

```jhyy
// compiler/src0/target/abi_amd64_win.jhyy

// emit QBE function header (signature string)
// is_sret = 1 时前置隐藏 l %ret 形参; param_qbe_types = 每参数 QBE type letter (w/l/s/d/b)
// emit: "export function <ret_qbe_type> $name(<params>) {\n@start\n"
fn abi_win_emit_function_header(
    ir: *IRBuf,
    name: *u8,
    n_params: i32,
    is_sret: i32,
    ret_qbe_type: *u8,
    param_qbe_types: *u8,    // packed QBE type letters
    param_names: *u8          // packed SSA names (e.g. "%t3 %t5 %t7")
) -> i32

// emit QBE return instruction (ret path)
// is_sret = 1 时先 copy_struct 到 sret slot 再 bare ret
// 否则 emit "    ret %val\n"
fn abi_win_emit_return(
    ir: *IRBuf,
    val: *u8,           // SSA name (或 "" for void)
    is_sret: i32,
    ret_type: *Type,
    sret_slot: *u8      // SSA name of alloc8 slot for sret struct (only when is_sret=1)
) -> i32

// emit call site args prelude (rcx/rdx/r8/r9 placement 由 QBE 处理,jhyy 只 build args 数组)
// 对 struct 实参: alloc8 slot + cg_copy_struct + 传 slot 地址 (隐式 slot alloc + copy)
fn abi_win_emit_call_prelude(
    ir: *IRBuf,
    args_arr: *IRVal,   // in/out, struct arg slot 替换原 val
    n_args: i32
) -> i32

// classify QBE type letter for a JHYY type
// i32/i64/ptr → 'w'/'l'; struct > 8 bytes → 'l' (slot ptr); struct ≤ 8 → 'l' (slot ptr); array → 'l'
fn abi_win_classify_arg(t: *Type) -> i32  // returns QBE type letter as i32

// emit struct arg copy-to-slot pattern
// alloc8 <size>; cg_copy_struct from <arg_val> to <slot>; replace arg val with slot SSA name
fn abi_win_emit_struct_arg_slot(
    ir: *IRBuf,
    arg_val: *u8,       // in: original struct SSA name; out: slot SSA name
    struct_type: *Type
) -> i32
```

**jhyy enum idiom**:`abi_win_classify_arg` 返回 i32 (`'w'`=119, `'l'`=108, `'s'`=115, `'d'`=100, `'b'`=98 作为 QBE type letter)。**不用 algebraic enum**(jhyy 不支持,per Explore agent Pattern A 建议,跟 `target_dispatch.jhyy:26-28` mimic 一致)。

#### 1.1.2 codegen.jhyy 抽离调用

- `compiler/src0/codegen.jhyy`（改）：
  - `import target/abi_amd64_win;`
  - `cg_func`（`:3674-3727` ~198 LOC）— 删 inline signature emit,改调 `abi_win_emit_function_header`
  - `cg_expr NODE_RETURN`（`:1474-1586` ~110 LOC）— 删 inline ret 路径,改调 `abi_win_emit_return`
  - `cg_expr NODE_CALL`（`:2318-2516` ~198 LOC）— 删 inline args arena,改调 `abi_win_emit_call_prelude`
  - `cg_expr NODE_QUALIFIED_CALL`（`:2518-2689` ~170 LOC）— 同样改调
  - `cg_copy_struct`（`:602-676` ~75 LOC）— 改调 `abi_win_emit_struct_arg_slot`(sret + struct arg 共用)
  - `cg_convert_arg`（`:752-897` ~146 LOC）— type coercion 是 ABI 无关的指令选择,保留 codegen.jhyy(非 ABI 范畴);但**保留原位置,不抽**
- codegen.jhyy 净减 ≥ 800 LOC

## Stage 2: abi_amd64_win_freestanding 新建

### 2.1 范围

#### 2.1.1 新建 `abi_amd64_win_freestanding.jhyy`(复用 + 增量)

```jhyy
// compiler/src0/target/abi_amd64_win_freestanding.jhyy
//
// UEFI 风格 ABI (per D-GUI-12): ABI = MS x64 (复用 abi_amd64_win.*)
// 区别: emit entry point + no_crt_init (no-op, 真验证推 v2.3.0 lld-link 阶段)

fn abi_fs_emit_function_header(...) -> i32 {
    return abi_win_emit_function_header(...);  // ABI 相同
}

fn abi_fs_emit_return(...) -> i32 {
    return abi_win_emit_return(...);  // ABI 相同
}

fn abi_fs_emit_call_prelude(...) -> i32 {
    return abi_win_emit_call_prelude(...);  // ABI 相同
}

// emit UEFI entry point section + global symbol
// emit "section \".text.boot\"\nexport $entry_name\nsection \".text\"\n"
fn abi_fs_emit_entry_point(ir: *IRBuf, entry_name: *u8) -> i32

// no-op (v2.1.0): codegen 不会主动 emit ucrt 引用 (v1.x 真自举后无 ucrt symbols)
// 实际验证推到 v2.3.0 lld-link hello-freestanding.efi 阶段
fn abi_fs_no_crt_init() -> i32 {
    return 0;  // no-op; QBE amd64 backend 决定实际 ucrt 引用
}
```

#### 2.1.2 codegen.jhyy `cg_module` 的 freestanding block 改实

v2.0.0 留的 `if (t == TARGET_AMD64_WIN_FREESTANDING) { stderr; return 0; }` inline block(v2.0.0 ship 时已经移除 stub 函数命名,直接是 inline 块)改成:

```jhyy
if t == TARGET_AMD64_WIN_FREESTANDING() {
    // 1. emit entry point section (efi_main, v2.3.0 build_efi.sh 改通用 entry name)
    abi_fs_emit_entry_point(ir, "efi_main" as *u8);

    // 2. 复用 abi_amd64_win 的 signature shaping (ABI 相同)
    //    [遍历 fn_decl 调 abi_fs_emit_function_header + emit_func_body + abi_fs_emit_return]

    // 3. no ucrt init (no-op, v2.1.0)
    abi_fs_no_crt_init();

    return 1 as i32;  // success
}
```

### 2.3 验收
1. **freestanding 编通**:`jhyy --target=amd64_win_freestanding -c hello.jhyy -o hello-freestanding.obj` → `file hello-freestanding.obj` 输出 `PE/COFF x86-64`(per v2.3.0 OVMF boot 链路前置)
2. **.il 含 entry point section**:`jhyy_get_il` 拿 .il,grep `section ".text.boot"` + `export $efi_main` 命中

新建 `compiler/tests/examples/abi_amd64_win/abi_test.jhyy`:

```jhyy
// test_signature_sret — struct 返回的 QBE IL 含 `l %ret` 隐藏指针
struct Point { x: i32, y: i32 }
fn make_point(x: i32, y: i32) -> Point { return Point { x: x, y: y }; }

fn test_signature_sret() -> i32 {
    // 验证 .il 含 "export function ... (l %ret, w %x, w %y)"
    let p = make_point(3, 4);
    return p.x + p.y;
}

// test_signature_struct_arg — struct 实参被 copy 到 alloc8 slot
fn take_point(p: Point) -> i32 { return p.x + p.y; }

fn test_signature_struct_arg() -> i32 {
    let p = make_point(5, 6);
    return take_point(p);
}

// test_signature_multi_arg — 4 参数全走 register (QBE type letter)
fn add4(a: i64, b: i64, c: i64, d: i64) -> i64 { return a+b+c+d; }

fn test_signature_multi_arg() -> i32 {
    let r = add4(1, 2, 3, 4);
    return r as i32;
}
```

**测试方法**(unit test 不能直接 grep .s,只能验 QBE signature 字符串):
- 跑 `jhyy abi_test.jhyy -c -o abi_test.obj`(只编 .obj)
- `jhyy_get_il abi_test.jhyy` 拿 .il 文本(或用 jhyy 编时把 .il 输出到 known path)
- grep signature 字符串:
  - `test_signature_sret`: `export function ... (l %ret, w %x, w %y)` 含 `l %ret`
  - `test_signature_struct_arg`: `export function ... (l %p)` 或 `export function ... take_point(...)` 调用前 `alloc8 <size>`
  - `test_signature_multi_arg`: `export function ... add4(w %a, w %b, w %c, w %d)` 4 个 `w` 参数

> **jhyy-side 验 x64 register placement 不可行**(grep .s 在 jhyy 内部不支持;QBE 输出 register 分配是 opaque);**走 regress.py + byte-equal 兜底**(行为不变即可)。
