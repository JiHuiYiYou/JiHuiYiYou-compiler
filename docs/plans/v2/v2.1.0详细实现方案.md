# v2.1.0 详细实现方案: ABI 抽离（Windows x64 + UEFI 风格）

## Overview

v2.1.0 在 v2.0.0（target dispatcher 起步）基础上抽离 Windows x64 ABI + UEFI 风格 ABI 到独立文件，让 codegen.jhyy 仅做 dispatch + 指令选择，不直接接触 ABI 细节。

**v2.0.0 阶段已有的相关基础**（v2.1.0 直接复用）:
- `target_dispatch.jhyy`：Target enum + `target_parse` + `target_default` + `target_name`
- `codegen.jhyy` 主入口加 target 参数（Amd64Win 完整路径不变 + 其他 stub）
- `main.jhyy` 加 `--target=<triple>` CLI flag
- v1.x 真自举闭环 + Stage 2 N=3 byte-equal baseline

**v2.1.0 范围**:
- 3 个 stage 串行（abi_amd64_win 抽离 → abi_amd64_win_freestanding 新建 → codegen 抽离调用 + ABI 单元测试）
- 不并行 v3.0 3a-3f（per `v2-v3-parallel-sprint-plan.md § 5.1` 路径 A）
- v3.0 6 特性等 v2.0 阶段 ship 后才启动

## Stage 1 详细设计: abi_amd64_win 抽离

### 1.1 emit_win_func_prologue 实现细节

#### 1.1.1 函数签名

```jhyy
// compiler/src0/target/abi_amd64_win.jhyy

fn emit_win_func_prologue(ir: *IRBuf, name: [*]u8, n_args: i32, locals_size: i64)
```

**参数说明**:
- `ir`：IR 缓冲
- `name`：函数名（用于 emit 调试注释）
- `n_args`：参数个数（用于判断是否需要 shadow space）
- `locals_size`：局部变量总大小（含 shadow space）

#### 1.1.2 完整实现

```jhyy
fn emit_win_func_prologue(ir: *IRBuf, name: [*]u8, n_args: i32, locals_size: i64) {
    // 注释: 函数名
    ir_emit_comment(ir, "function %s", name);

    // push rbp
    ir_emit(ir, "    %%t1 =l alloc4 8\n");
    ir_emit(ir, "    store %%rbp, %%t1\n");

    // mov rbp, rsp
    ir_emit(ir, "    %%rbp =l copy %%rsp\n");

    // sub rsp, locals_size（包含 shadow space）
    let total_size = locals_size;  // shadow space 已包含在 locals_size
    if (total_size > 0) {
        ir_emit(ir, "    %%rsp =l sub %%rsp, %lld\n", total_size);
    }

    // 保存非易失寄存器（如果函数用）
    // rbx/rsi/rdi/r12/r13/r14/r15
    // (v2.1.0 简化: 只在需要时保存; v2.x 中期升级)
}
```

#### 1.1.3 关键决策

| 决策 | 选 | 理由 |
|------|-----|------|
| shadow space 大小 | 32 字节 | per MS x64 强制 |
| locals_size 含 shadow space | ✅ | 调用方在 codegen 计算 locals 时已包含 |
| 是否保存非易失寄存器 | 仅需要时 | v2.1.0 简化；v2.x 中期升级完整 prologue |
| prologue 注释 | ✅ | 调试用，方便 .s diff |

```jhyy
    // mov rsp, rbp
    ir_emit(ir, "    %%rsp =l copy %%rbp\n");

    // pop rbp
    ir_emit(ir, "    %%rbp =l load %%t1\n");
```

### 1.3 emit_win_call 实现细节

```jhyy
fn emit_win_call(ir: *IRBuf, name: [*]u8, args: [*]IRVal, n_args: i32) {
    // 1. 把参数按 ARG_REGS_WIN 顺序放到 rcx/rdx/r8/r9
    let arg_regs: [i32; 4] = [0, 2, 8, 9];  // rcx/rdx/r8/r9
    let i = 0;
    while (i < n_args && i < 4) {
        ir_emit(ir, "    %%r%d =l copy %%t%d\n", arg_regs[i], args[i].id);
        i = i + 1;
    }

    // 2. 第 5+ 个参数走栈（v2.1.0 简化: 不实现, 报错）
    if (n_args > 4) {
        fatal("function %s has %d args, but amd64_win only supports 4 register args (v2.1.0 limit)", name, n_args);
    }

    // 3. call <name>
    ir_emit(ir, "    call $%s\n", name);
}
```

### 1.4 classify_arg_win 实现细节

```jhyy
fn classify_arg_win(t: *Type) -> ArgClass {
    // per v0.4 ABI:
    // - i32/i64/ptr → register
    // - struct > 8 bytes → sret (隐藏指针 rcx)
    // - struct ≤ 8 bytes → register (按值)
    // - array → sret (按指针)

    if (t->kind == TYPE_KIND_I32 || t->kind == TYPE_KIND_I64 || t->kind == TYPE_KIND_PTR) {
        return ArgClass::Reg(0);  // idx 由调用方分配
    }

    if (t->kind == TYPE_KIND_STRUCT) {
        if (type_size(t) <= 8) {
            return ArgClass::Reg(0);  // 按值传
        } else {
            return ArgClass::Sret;  // 隐藏指针
        }
    }

    if (t->kind == TYPE_KIND_ARRAY) {
        return ArgClass::Sret;  // 数组永远走 sret
    }

    fatal("classify_arg_win: unsupported type kind %d", t->kind);
}
```

### 1.5 codegen.jhyy 抽离调用详细改法

#### 1.5.1 当前 codegen.jhyy 的 emit 函数

```jhyy
// 当前（v2.0.0）:
fn emit_amd64_win(m: *Module, ir: *IRBuf) {
    // 函数 prologue（直接 emit QBE IL）
    for fn_decl in m->funcs {
        emit_func_prologue_inline(ir, fn_decl);  // 直接写 prologue
        emit_func_body(ir, fn_decl);
        emit_func_epilogue_inline(ir, fn_decl);  // 直接写 epilogue
    }

    // 全局变量 / data 段
    for var_decl in m->vars {
        emit_data_var_inline(ir, var_decl);  // 直接写 data 段
    }
}
```

#### 1.5.2 抽离后

```jhyy
// v2.1.0 后:
fn emit_amd64_win(m: *Module, ir: *IRBuf) {
    for fn_decl in m->funcs {
        // prologue
        let locals_size = compute_locals_size(fn_decl) + SHADOW_SPACE_SIZE;
        emit_win_func_prologue(ir, fn_decl->name, fn_decl->n_args, locals_size);

        // 函数体（保留 v1.x 实现）
        emit_func_body(ir, fn_decl);

        // epilogue
        emit_win_func_epilogue(ir, locals_size);
    }

    for var_decl in m->vars {
        emit_data_var(ir, var_decl);  // data 段保留（不抽离）
    }
}
```

#### 1.5.3 关键决策

- **data 段保留 codegen.jhyy**：data 段不涉及 ABI 复杂逻辑（仅 emit 字节），保留 codegen.jhyy 避免过度抽离
- **compute_locals_size 函数**：v2.1.0 新增，在 codegen.jhyy 计算 locals 总大小（含 shadow space）
- **函数体保留 codegen.jhyy**：函数体内部指令选择（assign / binop / call 等）保留，不抽离

```jhyy
fn emit_freestanding_no_crt_init(ir: *IRBuf) {
    // 不 emit __security_cookie / __report_rangecheckfailure / __chkstk 等 ucrt 引用
    // 不 emit __main / __do_global_ctors 等 vcruntime 引用
    //
    // 实现: codegen 主入口在 freestanding target 下不 emit 这些 symbols
    //       (v2.1.0 简化: 仅在注释中标明 "freestanding mode: no ucrt/vcruntime");
```

### 2.3 emit_freestanding_entry 实现细节

```jhyy
fn emit_freestanding_entry(ir: *IRBuf, entry_name: [*]u8) {
    // emit `.text.boot:` section
    ir_emit(ir, "section \".text.boot\"\n");

    // emit `global <entry_name>`
    ir_emit(ir, "export $%s\n", entry_name);

    // emit `.text:` section 恢复
    ir_emit(ir, "section \".text\"\n");
}
```

### 2.5 codegen.jhyy emit_amd64_win_freestanding 改实

```jhyy
// v2.1.0 后:
fn emit_amd64_win_freestanding(m: *Module, ir: *IRBuf) {
    // 1. emit entry point section (efi_main / _start)
    emit_freestanding_entry(ir, "efi_main");  // v2.3.0 改通用 entry name

    // 2. 复用 abi_amd64_win 的 prologue/epilogue/call (因为 ABI 相同)
    for fn_decl in m->funcs {
        let locals_size = compute_locals_size(fn_decl) + SHADOW_SPACE_SIZE;
        emit_freestanding_func_prologue(ir, fn_decl->name, fn_decl->n_args, locals_size);
        emit_func_body(ir, fn_decl);
        emit_freestanding_func_epilogue(ir, locals_size);
    }

    // 3. no ucrt init (验证: nm 输出空)
    emit_freestanding_no_crt_init(ir);

    // 4. data 段（保留 codegen.jhyy, 不抽离）
    for var_decl in m->vars {
        emit_data_var(ir, var_decl);
    }
}
```

```jhyy
fn emit_amd64_win(m: *Module, ir: *IRBuf) {
    for fn_decl in m->funcs {
```
