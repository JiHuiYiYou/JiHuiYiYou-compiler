# v2.1.0 详细实现方案: QBE-level ABI 抽离（Windows x64 + UEFI 风格）

## Overview

v2.1.0 在 v2.0.0（target dispatcher 起步）基础上抽离 **QBE-level** ABI shaping 到独立文件，让 codegen.jhyy 仅做 dispatch + 指令选择，不直接接触 ABI 细节。

**重要 scope 修订**(2026-09-02 doc audit):JHYY 编译器 emit QBE IL,**所有 x64 细节由 QBE 处理**(register 分配、shadow space、frame layout、callee-saved)。jhyy-side 能抽的实际是 QBE-level ABI shaping:

1. **QBE function signature 构造** — `export function w $name(l %ret, w %a, w %b) {` + `@start` 标签
2. **sret 预处理** — struct 返回 → 隐藏 `l %ret` 前置 + sret slot alloc + 返回前 copy_struct
3. **struct-arg 拷贝到 slot** — struct 实参 → `alloc8 <bytes>` + `cg_copy_struct` 拷贝 + 传 slot 地址
4. **type coercion at call site** — `cg_convert_arg` (dtosi/stosi/extsw/extuw/sltof/truncd QBE ops)

**不抽**(grep 验证 0 命中,不存在或 QBE 内部):
- `compute_locals_size` / `frame_size` / `locals_size`
- `emit_win_func_prologue` / `_epilogue`(无 raw x64 emit)
- shadow space 32 字节(QBE 处理)
- ARG_REGS_WIN register 数组(QBE 处理)

**v2.0.0 阶段已有的相关基础**(v2.1.0 直接复用):
- `target_dispatch.jhyy`：Target enum mimic + `target_parse` + `target_default` + `target_name`
- `codegen.jhyy` 主入口加 target 参数(Amd64Win 完整路径不变 + 其他 stub)
- `main.jhyy` 加 `--target=<triple>` CLI flag
- v1.x 真自举闭环 + Stage 2 N=4 byte-equal baseline
- v2.0.0 actual regress baseline

**v2.1.0 范围**:
- 4 个 stage 串行(plan doc reframe → abi_amd64_win 抽离 → abi_amd64_win_freestanding 新建 → codegen 抽离调用 + ABI signature-level tests)
- 不并行 v3.0 3a-3f(per `v2-v3-parallel-sprint-plan.md § 5.1` 路径 A)
- v3.0 6 特性等 v2.0 阶段 ship 后才启动

## Stage 1 详细设计: abi_amd64_win 抽离

### 1.1 abi_win_emit_function_header 实现细节

#### 1.1.1 函数签名

```jhyy
// compiler/src0/target/abi_amd64_win.jhyy
//
// emit QBE function header (signature string)
// is_sret = 1 时前置隐藏 l %ret 形参
//
// 示例:
//   非 sret: "export function w $add(w %t3, w %t4, w %t5, w %t6) {\n@start\n"
//   sret:    "export function w $make_point(l %ret, w %x, w %y) {\n@start\n"
//
// Returns: 1 = success, 0 = fatal (jhyy convention)
fn abi_win_emit_function_header(
    ir: *IRBuf,
    name: *u8,                // 函数名 (e.g. "add", "make_point")
    n_params: i32,            // 参数个数 (不含隐藏 sret slot)
    is_sret: i32,             // 0 or 1
    ret_qbe_type: *u8,        // QBE type letter ('w'/'l'/...)
    param_qbe_types: *u8,     // packed QBE type letters (e.g. "www" for 3 i32)
    param_names: *u8          // packed SSA names (e.g. "%t3 %t5 %t7")
) -> i32
```

#### 1.1.2 完整实现

```jhyy
fn abi_win_emit_function_header(
    ir: *IRBuf,
    name: *u8,
    n_params: i32,
    is_sret: i32,
    ret_qbe_type: *u8,
    param_qbe_types: *u8,
    param_names: *u8
) -> i32 {
    // 1. emit "export function <ret> $name("
    let prefix = "export function " as *u8;
    append_to_ir_buf(ir, prefix);
    append_to_ir_buf(ir, ret_qbe_type);
    let space = " " as *u8;
    append_to_ir_buf(ir, space);
    let dollar = "$" as *u8;
    append_to_ir_buf(ir, dollar);
    append_to_ir_buf(ir, name);

    // 2. emit "("<params>" ) {\n@start\n"
    let lparen = " (" as *u8;
    append_to_ir_buf(ir, lparen);

    // 2a. if sret, 前置 "l %ret, "
    if is_sret == (1 as i32) {
        let sret_prefix = "l %ret, " as *u8;
        append_to_ir_buf(ir, sret_prefix);
    }

    // 2b. emit "<param_qbe_types> <param_names>" 用 ", " 分隔
    let mut i: i32 = 0 as i32;
    while i < n_params {
        // emit type letter
        let type_letter = (param_qbe_types as i64 + (i as i64)) as *u8;
        append_to_ir_buf(ir, type_letter);
        let space2 = " " as *u8;
        append_to_ir_buf(ir, space2);
        // emit param name (从 packed string 解析; v2.1.0 简化: 假设 param_names 是单个 SSA name)
        // 真实实现: 解析 packed SSA names 用空格/逗号分隔
        let param_name = /* ... 解析 packed names ... */;
        append_to_ir_buf(ir, param_name);
        if i < n_params - (1 as i32) {
            let comma = ", " as *u8;
            append_to_ir_buf(ir, comma);
        }
        i = i + (1 as i32);
    }

    // 3. emit ") {\n@start\n"
    let suffix = ") {\n@start\n" as *u8;
    append_to_ir_buf(ir, suffix);

    return 1 as i32;
}
```

#### 1.1.3 关键决策

| 决策 | 选 | 理由 |
|------|-----|------|
| `param_qbe_types` packed string vs array | packed string(i64 长度) | jhyy 数组 `[*]u8` 已用;packed 简洁,O(N) 解析 |
| `param_names` 同 packed | packed string | 跟 types 同步;v2.1.0 简化:假设 single SSA name per param |
| sret hidden param name | `%ret`(固定) | QBE 规范允许任意名;`%ret` 直观 |
| 函数体 emit 留在 codegen.jhyy | ✅ | `emit_func_body` 是指令选择,非 ABI 范畴 |

### 1.2 abi_win_emit_return 实现细节

#### 1.2.1 函数签名

```jhyy
// emit QBE return instruction (ret path)
// is_sret = 1 时先 copy_struct 到 sret slot 再 bare ret
//
// 示例:
//   non-sret void:  "    ret\n"
//   non-sret value: "    ret %t42\n"
//   sret:           cg_copy_struct(%val, %sret_slot); "    ret\n"
fn abi_win_emit_return(
    ir: *IRBuf,
    val: *u8,           // SSA name (or "" for void)
    is_sret: i32,
    ret_type: *Type,
    sret_slot: *u8      // SSA name of alloc8 slot (only when is_sret=1)
) -> i32
```

#### 1.2.2 完整实现

```jhyy
fn abi_win_emit_return(
    ir: *IRBuf,
    val: *u8,
    is_sret: i32,
    ret_type: *Type,
    sret_slot: *u8
) -> i32 {
    if is_sret == (1 as i32) {
        // 1. cg_copy_struct from val to sret_slot
        cg_copy_struct(ir, val, sret_slot, ret_type);
        // 2. emit bare ret
        let bare_ret = "    ret\n" as *u8;
        append_to_ir_buf(ir, bare_ret);
    } else if val_is_void(val) {
        // 3. void ret
        let bare_ret = "    ret\n" as *u8;
        append_to_ir_buf(ir, bare_ret);
    } else {
        // 4. value ret
        let prefix = "    ret " as *u8;
        append_to_ir_buf(ir, prefix);
        append_to_ir_buf(ir, val);
        let newline = "\n" as *u8;
        append_to_ir_buf(ir, newline);
    }
    return 1 as i32;
}
```

### 1.3 abi_win_emit_call_prelude 实现细节

#### 1.3.1 函数签名

```jhyy
// emit call site args prelude (rcx/rdx/r8/r9 placement 由 QBE 处理,jhyy 只 build args 数组)
// 对 struct 实参: alloc8 slot + cg_copy_struct + 传 slot 地址
//
// 示例: call $add4(w %t3, w %t4, w %t5, w %t6) → 4 个 i64 参数
// QBE 自动分配到 rcx/rdx/r8/r9
//
// 示例: call $take_point(l %slot) → struct Point 实参被 copy 到 alloc8 slot
fn abi_win_emit_call_prelude(
    ir: *IRBuf,
    args_arr: *IRVal,   // in/out: struct arg slot 替换原 val
    n_args: i32
) -> i32
```

#### 1.3.2 完整实现

```jhyy
fn abi_win_emit_call_prelude(
    ir: *IRBuf,
    args_arr: *IRVal,
    n_args: i32
) -> i32 {
    let mut i: i32 = 0 as i32;
    while i < n_args {
        // 1. check arg type
        let arg_type = arg_get_type(args_arr, i);
        let kind = type_kind(arg_type);

        if kind == TYPE_KIND_STRUCT {
            // 2. struct arg: alloc8 slot + cg_copy_struct + 替换 val
            let arg_val = arg_get_val(args_arr, i);
            let struct_size = type_size(arg_type);
            let slot_ssa = alloc_temp_ssa();
            // emit "    <slot_ssa> =l alloc8 <struct_size>\n"
            emit_alloc8(ir, slot_ssa, struct_size);
            // cg_copy_struct from arg_val to slot_ssa
            cg_copy_struct(ir, arg_val, slot_ssa, arg_type);
            // replace args_arr[i].val with slot_ssa (so caller passes slot address)
            arg_set_val(args_arr, i, slot_ssa);
        } else if kind == TYPE_KIND_ARRAY {
            // 3. array arg: 同样走 slot (array 永走指针)
            let arg_val = arg_get_val(args_arr, i);
            let array_size = type_size(arg_type);
            let slot_ssa = alloc_temp_ssa();
            emit_alloc8(ir, slot_ssa, array_size);
            cg_copy_struct(ir, arg_val, slot_ssa, arg_type);
            arg_set_val(args_arr, i, slot_ssa);
        }
        // 4. primitive (i32/i64/ptr): pass as-is, type coercion 在 cg_convert_arg(保留 codegen.jhyy)
        i = i + (1 as i32);
    }
    return 1 as i32;
}
```

### 1.4 abi_win_classify_arg 实现细节

#### 1.4.1 函数签名

```jhyy
// classify QBE type letter for a JHYY type
//
// Returns: i32 = QBE type letter as ASCII ('w'=119, 'l'=108, 's'=115, 'd'=100, 'b'=98)
//
// per jhyy ABI:
//   i32 → 'w' (word)
//   i64 → 'l' (long / 8 bytes)
//   ptr → 'l' (8 bytes on x64)
//   struct → 'l' (slot pointer, alloc8'd and passed by ref)
//   array → 'l' (always slot pointer)
fn abi_win_classify_arg(t: *Type) -> i32
```

#### 1.4.2 完整实现

```jhyy
fn abi_win_classify_arg(t: *Type) -> i32 {
    let kind = (*t).kind;

    if kind == TYPE_KIND_I32 {
        return 119 as i32;  // 'w'
    }
    if kind == TYPE_KIND_I64 {
        return 108 as i32;  // 'l'
    }
    if kind == TYPE_KIND_PTR {
        return 108 as i32;  // 'l'
    }
    if kind == TYPE_KIND_STRUCT {
        return 108 as i32;  // 'l' (slot ptr)
    }
    if kind == TYPE_KIND_ARRAY {
        return 108 as i32;  // 'l' (slot ptr)
    }

    // fatal: unsupported type
    jh_fputs_stderr("abi_win_classify_arg: unsupported type kind " as *u8);
    print_int(kind);
    jh_fputs_stderr("\n" as *u8);
    return 0 as i32;
}
```

#### 1.5.1 函数签名

```jhyy
// emit struct arg copy-to-slot pattern (extract from cg_copy_struct for sret/struct arg reuse)
// alloc8 <size>; cg_copy_struct from <arg_val> to <slot>; return slot SSA name
fn abi_win_emit_struct_arg_slot(
    ir: *IRBuf,
    arg_val: *u8,       // in: original struct SSA name
    struct_type: *Type  // in: struct type for size + field layout
) -> *u8              // out: slot SSA name
```

#### 1.5.2 完整实现

```jhyy
fn abi_win_emit_struct_arg_slot(
    ir: *IRBuf,
    arg_val: *u8,
    struct_type: *Type
) -> *u8 {
    // 1. alloc temp SSA name
    let slot_ssa = alloc_temp_ssa();

    // 2. emit "    <slot_ssa> =l alloc8 <struct_size>\n"
    let struct_size = type_size(struct_type);
    emit_alloc8(ir, slot_ssa, struct_size);

    // 3. cg_copy_struct from arg_val to slot_ssa
    cg_copy_struct(ir, arg_val, slot_ssa, struct_type);

    return slot_ssa;
}
```

### 2.2 abi_fs_emit_entry_point 实现

```jhyy
// emit UEFI entry point section + global symbol
//
// emit "section \".text.boot\"\nexport $entry_name\nsection \".text\"\n"
//
// v2.1.0 hardcode "efi_main" (per v2.3.0 build_efi.sh); v2.2.0+ 通用 entry name CLI flag
fn abi_fs_emit_entry_point(ir: *IRBuf, entry_name: *u8) -> i32 {
    let sec_boot = "section \".text.boot\"\n" as *u8;
    append_to_ir_buf(ir, sec_boot);
    let export_prefix = "export $" as *u8;
    append_to_ir_buf(ir, export_prefix);
    append_to_ir_buf(ir, entry_name);
    let newline = "\n" as *u8;
    append_to_ir_buf(ir, newline);
    let sec_text = "section \".text\"\n" as *u8;
    append_to_ir_buf(ir, sec_text);
    return 1 as i32;
}
```

> **v2.1.0 简化理由**:codegen.jhyy 在 v1.x 真自举后不会主动 emit `__security_cookie` / `__report_rangecheckfailure` / `__chkstk` / `__main` 等 ucrt 引用。但 QBE amd64 backend 在生成 .s 时**可能**会 emit 这些 symbols 的引用(取决于 QBE 版本)。v2.1.0 不验证(no-op),推到 v2.3.0 lld-link .efi 阶段真验。

### 2.4 codegen.jhyy `cg_module` freestanding block 改实

v2.0.0 留的 inline block(无函数命名,直接是 `if (t == ...) { stderr; return 0; }`)改成:

```jhyy
// v2.1.0 后:
fn cg_module(ir: *IRBuf, module: *Node, t: i32) -> i32 {
    if module == (0 as *Node) { return 0; }

    // freestanding: emit entry point + 复用 MS x64 ABI
    if t == TARGET_AMD64_WIN_FREESTANDING() {
        abi_fs_emit_entry_point(ir, "efi_main" as *u8);
        // [遍历 fn_decl 调 abi_fs_emit_function_header + emit_func_body + abi_fs_emit_return]
        // ...
        abi_fs_no_crt_init();
        return 1 as i32;  // success (jhyy convention)
    }

    // sysv stub (v2.1.0 不实现)
    if t == TARGET_AMD64_SYSV_STUB() {
        jh_fputs_stderr("amd64_sysv target: 实现留 v2.x M2\n" as *u8);
        return 0 as i32;
    }

    // Amd64Win (默认): 走 Stage 1 抽离后的 abi_win_* 函数
    // [原有 cg_func + cg_copy_struct + cg_convert_arg 走 abi_win_* 调用]
    // ...
    return 1 as i32;
}
```

```jhyy
fn cg_module(ir: *IRBuf, module: *Node, t: i32) -> i32 {
```
