#### 1.1.1 CGContext 加 module globals dict

**当前** (`compiler/src/codegen.c` line ~50-60):
```c

    /* ... 现有 fields ... */
    int last_dbg_line;       /* v1.4.2 */
    int dbg_file_emitted;    /* v1.4.2 */
```

**改动后**:
```c
/* Module-level globals dict: Sym* → (QBE $name string, QBE type letter).
   Populated by cg_module pass A on NODE_LET; looked up by cg_find_local
   when local miss falls through. Lifetime = whole module (one dict per
   cg_module call, shared across all cg_func calls). */
typedef struct CGModGlobal {
    Sym *sym;
    char *qbe_name;     /* "$<name>" — QBE global identifier */
    char qbe_type;      /* 'w'/'l'/'b' — QBE type letter for load/store size */
} CGModGlobal;

typedef struct {
    /* ... 现有 fields ... */
    int last_dbg_line;
    int dbg_file_emitted;
    /* v1.4.6: W-017 真修 — module-level globals (cross-fn lifetime) */
    CGModGlobal *mod_globals;
    size_t n_mod_globals;
    size_t cap_mod_globals;
    Arena *arena;       /* owned by IRBuf, 借用 — 用于 CGModGlobal.qbe_name strdup */
} CGContext;
```

**jhyy-side mirror**: `compiler/src0/codegen.jhyy` CGContext 同步加 4 个 fields; CGCONTEXT_SIZE 112 → 168 (per `architecture.md` "CGContext layout 必须 mirror", Stage 1 byte-equal 守门)。

#### 1.1.2 Helper: `cg_mod_global_register` + `cg_mod_global_find`

```c
/* Append a module-level global. Returns the registered CGModGlobal*. */
static CGModGlobal *cg_mod_global_register(CGContext *cg, Sym *sym,
                                           const char *qbe_name, char qbe_type) {
    if (cg->n_mod_globals >= cg->cap_mod_globals) {
        size_t new_cap = cg->cap_mod_globals ? cg->cap_mod_globals * 2 : 16;
        CGModGlobal *ng = arena_alloc(cg->arena, new_cap * sizeof(CGModGlobal));
        if (cg->mod_globals && cg->n_mod_globals > 0)
            memcpy(ng, cg->mod_globals, cg->n_mod_globals * sizeof(CGModGlobal));
        cg->mod_globals = ng;
        cg->cap_mod_globals = new_cap;
    }
    CGModGlobal *g = &cg->mod_globals[cg->n_mod_globals++];
    g->sym = sym;
    /* arena_alloc 然后 strdup — qbe_name lifetime = module */
    size_t len = strlen(qbe_name) + 1;
    char *cn = arena_alloc(cg->arena, len);
    memcpy(cn, qbe_name, len);
    g->qbe_name = cn;
    g->qbe_type = qbe_type;
    return g;
}

/* Look up a module global by Sym*. Returns NULL if not registered. */
static CGModGlobal *cg_mod_global_find(CGContext *cg, Sym *sym) {
    for (size_t i = 0; i < cg->n_mod_globals; i++) {
        if (cg->mod_globals[i].sym == sym) return &cg->mod_globals[i];
    }
    return NULL;
}
```

**改动后** (追加 NODE_LET 分支):
```c
for (size_t i = 0; i < md->ndeccls; i++) {
    Node *decl = md->decls[i];
    if (decl->kind == NODE_CONST_DECL) {
        /* existing const decl emit ... */
    } else if (decl->kind == NODE_LET) {
        /* v1.4.6: W-017 真修 — 顶层 let / let mut emit module-level global */
        NodeLet *d = node_let_data(decl);
        Type *t = d->sym->type;
        if (!t) continue;
        char qbe_t = qbe_type_of(t);
        char qbe_name[256];
        snprintf(qbe_name, sizeof(qbe_name), "$%s", d->sym->name->name);
        /* Register in module globals dict */
        CGModGlobal *g = cg_mod_global_register(&cg /* module-level context */,
                                                 d->sym, qbe_name, qbe_t);
        /* Emit `data $name = { <init> }` for primitives + small aggregates.
           For structs/enums/arrays > 8B, emit `data $name` placeholder (zero-init)
           and rely on cg_stmt NODE_LET to emit copy-into-store at fn entry.
           Simplification: for v1.4.6, only emit zero-init data; the existing
           cg_stmt NODE_LET in main_jhyy will copy struct value into $name
           via cg_emit_store with the right size. */
        int size = (int)type_size(t);
        if (size < 4) size = 4;
        ir_emit_data(ir, "data %s = { %c 0 }\n", g->qbe_name,
                     size == 8 ? 'l' : 'w');
        /* If init expression is provided, store via cg_emit_store —
           but cg_emit_store requires a CGContext with locals. Instead,
           for module-level init, we use the same store path at first
           fn entry: when cg_find_local falls through to cg_mod_global_find,
           and the symbol is being read for the first time, emit a load
           preceded by a one-time initializer if d->init is non-NULL. */
        /* For v1.4.6 scope: only support primitive init expressions
           (NODE_INT_LIT / NODE_STR_LIT / NODE_CAST from int/ptr).
           Complex init (struct literal, fn call result) → defer to
           cg_stmt entry of first fn body that reads it. */
        if (d->init && d->init->kind == NODE_INT_LIT) {
            /* Overwrite the zero-init with actual value */
            ir_emit_data(ir, "data %s = { %c %lld }\n", g->qbe_name, qbe_t,
                         (long long)node_int_data(d->init));
        }
        /* Note: complex init deferred to v1.4.7+ — current jhyy-side
           main.jhyy top-level only has primitive inits (path ptrs). */
    }
}
```

### 1.3 `cg_find_local` 加 fallthrough

**当前** (`compiler/src/codegen.c:74-88`):
```c

        if (cg->locals[i].sym == sym) {
            *is_stack = cg->locals[i].is_stack;
            *out = cg->locals[i].val;
            return;
        }
    }
    /* not found → return sentinel IRVal{0} (kind=IRVAL_TEMP, id=0) */
    *out = (IRVal){0};
}
```

**改动后** (追加 fallthrough):
```c
static void cg_find_local(CGContext *cg, Sym *sym, int *is_stack, IRVal *out) {
    for (size_t i = 0; i < cg->nlocals; i++) {
        if (cg->locals[i].sym == sym) {
            *is_stack = cg->locals[i].is_stack;
            *out = cg->locals[i].val;
            return;
        }
    }
    /* v1.4.6: W-017 真修 — fallthrough to module globals dict */
    CGModGlobal *g = cg_mod_global_find(cg, sym);
    if (g) {
        /* module global: emit `load $name` into a new temp */
        IRVal tmp = ir_new_tmp(cg->ir, g->qbe_type);
        ir_emit(cg->ir, "    %%%c%d =%c load %s\n",
                g->qbe_type, tmp.id, g->qbe_type, g->qbe_name);
        *is_stack = 0; /* value, not address */
        *out = tmp;
        return;
    }
    /* not found anywhere → sentinel */
    *out = (IRVal){0};
}
```

**关键点**: 写 module global 时 (cg_stmt NODE_ASSIGN `x = ...` 路径), `cg_find_local` 返回 load 出的 value 是错的 — 需要 emit `store $name, new_value` 而不是 `store local_slot, new_value`。所以还需要 `cg_store_global` helper:

```c
static void cg_store_global(CGContext *cg, Sym *sym, IRVal value) {
    CGModGlobal *g = cg_mod_global_find(cg, sym);
    if (!g) return; /* not a global — silently skip */
    cg_emit_store(cg, /* type */ NULL, value, /* addr */ (IRVal){0} /* placeholder */);
    /* Actually we need a special path: emit `store new_value, $name`.
       Since cg_emit_store takes an IRVal addr, and QBE accepts
       `store src, $name` as a global store, we can use the qbe_name as
       a pseudo-IRVal. For simplicity: emit directly here. */
    ir_emit(cg->ir, "    store %%%c %s\n", g->qbe_type, g->qbe_name, value.id);
}
```

**注意**: jhyy-side 写 module global 路径需要新 helper — 不复用 cg_emit_store (那个是写 stack slot 的)。

## 子任务 2: W-019 真修 — 嵌套 struct field chain

### 2.1 根因诊断 (确诊)

**当前 `cg_expr` `case NODE_FIELD` 路径** (`compiler/src/codegen.c:1662-1687`):
```c
/* For struct value field access: the value IS the stack slot pointer */
IRVal base = {0};
cg_expr(cg, d->expr, &base);     /* 递归拿 base value */
Type *st = expr_type;
size_t offset = 0;
Type *field_type = NULL;
for (...) offset = ... ;         /* 找 field offset */
/* offset=0: addr = base (assume base is address) */
/* offset>0: addr = base + offset (assume base is address) */
IRVal addr = ...;
IRVal result = ir_new_tmp(cg->ir, qbe_type_of(field_type));
cg_emit_load(cg, result, field_type, addr);  /* loadw/loadl at addr */
```

**问题**: 当 `d->expr` 本身是 `NODE_FIELD` (e.g. `outer.inner`), `cg_expr(d->expr, &base)` 递归调用, 内部也走 NODE_FIELD 路径, 拿到的是**已经 load 出的 inner value** (一个 IRVal, 类型 = 'l' 但语义是 inner struct 的 value, 不是地址)。

随后 caller 把这个 value 当 address 用, `add offset` 拿到的地址偏移是相对 **value 的高字节**, 而 value 已经在寄存器里 (不是地址) → `loadsw` 拿到错字节 → "invalid type for first operand %t2 in loadsw" (QBE 看到的是一个 64-bit register 被当 32-bit address 解)。

### 2.2 修法: 三步 chain + `cg_alloc_temp_slot` helper

**新逻辑**:
```
base = cg_expr(d->expr);        /* step 1: 拿 .expr 的 value */
if base 不是 address (e.g. inner chain):
    slot = cg_alloc_temp_slot(base, size_of_type(d->expr.type));
    addr = slot;                /* slot 是真 address */
else:
    addr = base;                /* base 已经是 address */
addr += offset_of(field);       /* step 2: 加 field offset */
result = load(addr);            /* step 3: load field value */
```

**helper: `cg_alloc_temp_slot`**:
```c
/* Allocate a temporary stack slot, store `value` into it, return the slot
   address as an IRVal (kind=IRVAL_TEMP, type='l'). Used for nested struct
   field chain where intermediate values are not addresses. */
static IRVal cg_alloc_temp_slot(CGContext *cg, IRVal value, int size) {
    if (size < 4) size = 4;
    IRVal slot = ir_new_tmp(cg->ir, 'l');
    ir_emit_alloc(cg->ir, slot, size);
    ir_emit(cg->ir, "    %%%c%d =%c load %%%c%d\n" /* WRONG — placeholder */);
    /* Actually: emit `store value, slot` */
    cg_emit_store(cg, NULL, value, slot);
    return slot;
}
```

### 2.3 改 `cg_expr NODE_FIELD`

**当前路径** (line 1662-1687 简化):
```c
IRVal base = {0};
cg_expr(cg, d->expr, &base);
/* 假设 base 是 address — buggy for nested chain */
```

**改动后**:
```c
/* v1.4.6: W-019 真修 — 嵌套 struct field chain 区分 address-of vs value-of.
   If d->expr is itself a NODE_FIELD / NODE_INDEX (复合地址路径), the
   recursively-evaluated value is a value, not an address — allocate a
   temp slot to materialize the address. */
int base_is_value = (d->expr->kind == NODE_FIELD ||
                     d->expr->kind == NODE_INDEX ||
                     /* ... 其他 "复合地址" 节点 ... */);
IRVal base = {0};
cg_expr(cg, d->expr, &base);
Type *st = expr_type;
if (!st || st->kind != KIND_STRUCT) {
    *out = (IRVal){0}; return;
}
size_t offset = 0;
Type *field_type = NULL;
for (size_t i = 0; i < st->struct_type.nfields; i++) {
    if (strcmp(st->struct_type.fields[i].name->name, d->field) == 0) {
        offset = st->struct_type.fields[i].offset;
        field_type = st->struct_type.fields[i].type;
        break;
    }
}
IRVal addr;
if (base_is_value) {
    /* v1.4.6: W-019 真修 — allocate temp slot for nested field value */
    addr = cg_alloc_temp_slot(cg, base, (int)type_size(st));
    if (offset > 0) {
        IRVal adj = ir_new_tmp(cg->ir, 'l');
        ir_emit_binary(cg->ir, adj, "add", addr, ir_new_int((int64_t)offset));
        addr = adj;
    }
} else if (offset > 0) {
    addr = ir_new_tmp(cg->ir, 'l');
    ir_emit_binary(cg->ir, addr, "add", base, ir_new_int((int64_t)offset));
} else {
    addr = base;
}
IRVal result = ir_new_tmp(cg->ir, field_type ? qbe_type_of(field_type) : 'w');
cg_emit_load(cg, result, field_type, addr);
*out = result; return;
```

```markdown
4. `inc()` 调 2 次: 第一次返回 1 (load 0, store 1, load 1), 第二次返回 2
```
